28 // Vrai si le preset suit la charte v3 (metadata.charter_version "3...").
29 inline bool isCharterV3 (const nlohmann::json& doc)
30 {
31 if (! doc.contains ("metadata") || ! doc["metadata"].is_object())
32 return false;
33
34 const auto& m = doc["metadata"];
35 return m.contains ("charter_version") && m["charter_version"].is_string()
36 && m["charter_version"].get<std::string>().rfind ("3", 0) == 0;
37 }

39 // Vrai si les valeurs contiennent un paramètre apparu en v3 (engine, unison_voices, dx7_...) :
40 // seul indice quand le preset arrive sans metadata (entrée d'historique du backend).
41 // Un preset v1/v2 n'a que les 20 paramètres historiques ; un id inconnu ne compte pas.
42 inline bool hasV3Parameters (const std::map<juce::String, float>& values)
43 {
44 for (const auto& entry : values)
45 {
46 const auto id = entry.first.toStdString();
47
48 if (HarmoniaCharter::indexOf (id) >= 0
49 && std::find (HarmoniaCharter::v2Ids.begin(), HarmoniaCharter::v2Ids.end(), id)
50 == HarmoniaCharter::v2Ids.end())
51 return true;
52 }
53
54 return false;
55 }

57 // Valeurs v2 normalisées -> valeurs v3 normalisées, par les règles de la charte
58 // (HarmoniaCharter::upgradeFromV2Normalized) :
59 // - paramètre avec une règle v2 : converti, s'il vient d'un paramètre présent dans le preset
60 // - paramètre apparu en v3 (pas de règle) : sa valeur neutre, pour garder le son du preset
61 // - moteur : analogique (un preset v2 ne connaît pas le DX7)
62 inline std::map<juce::String, float> upgradeV2Values (const std::map<juce::String, float>& values)
63 {
64 std::array<float, HarmoniaCharter::v2Ids.size()> v2 {};
65 std::array<bool, HarmoniaCharter::v2Ids.size()> present {};
66
67 for (size_t i = 0; i < v2.size(); ++i)
68 {
69 const auto it = values.find (HarmoniaParams::toJuce (HarmoniaCharter::v2Ids[i]));
70 present[i] = it != values.end();
71 v2[i] = present[i] ? juce::jlimit (0.0f, 1.0f, it->second) : 0.0f;
72 }
73
74 // Aucun paramètre reconnu : rien à appliquer (l'appelant refuse le preset)
75 if (std::none_of (present.begin(), present.end(), [] (bool b) { return b; }))
76 return {};
77
78 std::map<juce::String, float> out;
79
80 for (size_t i = HarmoniaCharter::analogBegin; i < HarmoniaCharter::analogEnd; ++i)
81 {
82 const auto& spec = HarmoniaCharter::params[i];
83 const auto id = HarmoniaParams::toJuce (spec.id);
84
85 if (spec.v2.source < 0)
86 out[id] = HarmoniaCharter::toNormalized (spec, spec.defaultValue);
87 else if (present[(size_t) spec.v2.source])
88 out[id] = HarmoniaCharter::toNormalized (spec, HarmoniaCharter::upgradeFromV2Normalized (spec, v2));
89 // source absente du preset : le paramètre n'est pas touché
90 }
91
92 out[HarmoniaParams::IDs::engine] = 0.0f;
93 return out;
94 }

96 // Applique des valeurs normalisées { id -> [0..1] } sur l'APVTS.
97 // Seuls les ids présents dans `values` sont touchés : les ids inconnus (autre version du
98 // modèle / du plugin) sont ignorés, et les paramètres absents du preset gardent leur valeur actuelle.
99 // lockedIds : paramètres verrouillés, ils ne sont jamais modifiés.
100 // Renvoie le nombre de paramètres du preset reconnus (verrouillés compris).
101 // resetUnspecified : les paramètres absents de `values` (et non verrouillés) repartent à leur
102 // défaut. "engine" n'est jamais réinitialisé : c'est à l'appelant (ou au preset) de le fixer.
103 // Seuls les paramètres de la charte font partie d'un preset : les autres (volume de sortie,
104 // HarmoniaParams::Output) ne sont JAMAIS touchés, même si un fichier, l'IA ou l'historique les envoie.
105 inline bool isPresetParameter (const juce::String& id)
106 {
107 return HarmoniaCharter::indexOf (id.toStdString()) >= 0;
108 }

110 inline int applyValues (const std::map<juce::String, float>& values,
111 juce::AudioProcessorValueTreeState& apvts,
112 const juce::StringArray& lockedIds,
113 bool resetUnspecified = false)
114 {
115 int recognised = 0;
116 bool hasDx7 = false;
117
118 for (const auto& [id, value] : values)
119 {
120 if (! isPresetParameter (id) || apvts.getParameter (id) == nullptr)
121 continue;
122
123 ++recognised;
124 if (id.startsWith ("dx7_"))
125 hasDx7 = true;
126 }
127
128 if (recognised == 0)
129 return 0;
130
131 if (resetUnspecified)
132 {
133 for (const auto& spec : HarmoniaCharter::params)
134 {
135 const auto id = HarmoniaParams::toJuce (spec.id);
136
137 if (id == HarmoniaParams::IDs::engine // géré plus bas
138 || lockedIds.contains (id)
139 || values.count (id) > 0)
140 continue;
141
142 if (auto* p = apvts.getParameter (id))
143 {
144 p->beginChangeGesture();
145 p->setValueNotifyingHost (p->getDefaultValue());
146 p->endChangeGesture();
147 }
148 }
149
150 // Le preset ne dit pas quel moteur : on le déduit de ses paramètres.
151 if (values.count (HarmoniaParams::IDs::engine) == 0)
152 {
153 if (auto* e = apvts.getParameter (HarmoniaParams::IDs::engine))
154 {
155 e->beginChangeGesture();
156 e->setValueNotifyingHost (hasDx7 ? 1.0f : 0.0f);
157 e->endChangeGesture();
158 }
159 }
160 }
161
162 for (const auto& [id, value] : values)
163 {
164 auto* p = apvts.getParameter (id);
165 if (p == nullptr || lockedIds.contains (id) || ! isPresetParameter (id))
166 continue;
167
168 p->beginChangeGesture();
169 p->setValueNotifyingHost (juce::jlimit (0.0f, 1.0f, value));
170 p->endChangeGesture();
171 }
172
173 return recognised;
174 }

176 // Charge une entrée de l'historique du backend : un nom et { id -> valeur normalisée [0..1] },
177 // sans metadata. Le format (v3, ou v1/v2 à convertir) se lit sur les ids (hasV3Parameters).
178 // Les ids verrouillés ou inconnus sont ignorés ; une valeur hors de [0, 1] refuse l'entrée entière.
179 inline LoadResult loadFromParameters (const std::map<juce::String, float>& parameters,
180 const juce::String& presetName,
181 juce::AudioProcessorValueTreeState& apvts,
182 const juce::StringArray& lockedIds = {},
183 bool resetUnspecified = true)
184 {
185 LoadResult r;
186
187 // Valider avant de toucher au moindre paramètre
188 for (const auto& [id, value] : parameters)
189 {
190 if (! std::isfinite (value) || value < 0.0f || value > 1.0f)
191 {
192 r.errorMessage = "Invalid preset: value out of range";
193 return r;
194 }
195 }
196
197 const auto values = hasV3Parameters (parameters) ? parameters : upgradeV2Values (parameters);
198
199 if (applyValues (values, apvts, lockedIds, resetUnspecified) == 0)
200 {
201 r.errorMessage = "No matching parameters found in history entry";
202 return r;
203 }
204
205 r.success = true;
206 r.presetName = presetName.isNotEmpty() ? presetName : juce::String ("Preset");
207 return r;
208 }

210 // Charge un fichier hel.json (schéma : metadata + parameters{name:value} + values[])
211 // et applique les valeurs normalisées sur l'APVTS.
212 // Accepte deux formes :
213 // 1. parameters: { "osc_1_waveform": 0.5, ... } (recommandée, nombre de paramètres libre)
214 // 2. values: [0.5, 0.3, ...] (fallback ; ordre = HarmoniaParams::orderedIds())
215 // Un preset partiel est accepté : seuls les paramètres présents sont modifiés.
216 // Un preset v3 (charter_version "3...", ou des paramètres apparus en v3) s'applique tel quel ;
217 // un preset v1/v2 (modèle charter_v1, anciens fichiers, forme `values`) est converti.
218 //
219 // lockedIds : paramètres verrouillés, ils ne sont jamais modifiés.
220 inline LoadResult loadFromJsonString (const juce::String& jsonText,
221 juce::AudioProcessorValueTreeState& apvts,
222 const juce::StringArray& lockedIds = {},
223 bool resetUnspecified = false)
224 {
225 LoadResult r;
226
227 nlohmann::json doc;
228 try { doc = nlohmann::json::parse (jsonText.toStdString()); }
229 catch (const std::exception& e)
230 {
231 r.errorMessage = juce::String ("Invalid JSON: ") + e.what();
232 return r;
233 }
234
235 if (doc.contains ("metadata")
236 && doc["metadata"].is_object()
237 && doc["metadata"].contains ("name")
238 && doc["metadata"]["name"].is_string())
239 {
240 r.presetName = juce::String (doc["metadata"]["name"].get<std::string>());
241 }
242
243 // Valeurs normalisées du preset, par identifiant.
244 std::map<juce::String, float> values;
245
246 if (doc.contains ("parameters") && doc["parameters"].is_object())
247 {
248 for (auto it = doc["parameters"].begin(); it != doc["parameters"].end(); ++it)
249 if (it.value().is_number())
250 values[juce::String (it.key())] = (float) it.value().get<double>();
251 }
252 else if (doc.contains ("values") && doc["values"].is_array())
253 {
254 const auto& ids = HarmoniaParams::orderedIds();
255 const auto& arr = doc["values"];
256
257 // Le tableau doit correspondre exactement aux paramètres attendus.
258 if (arr.size() != ids.size())
259 {
260 r.errorMessage = "Invalid preset: incorrect number of values";
261 return r;
262 }
263
264 // Valider toutes les valeurs avant de modifier le moindre paramètre.
265 for (const auto& value : arr)
266 {
267 if (! value.is_number())
268 {
269 r.errorMessage = "Invalid preset: 'values' must contain only numbers";
270 return r;
271 }
272
273 const double number = value.get<double>();
274
275 if (! std::isfinite (number) || number < 0.0 || number > 1.0)
276 {
277 r.errorMessage = "Invalid preset: value out of range";
278 return r;
279 }
280 }
281
282 for (size_t i = 0; i < ids.size(); ++i)
283 values[ids[i]] = static_cast<float> (arr[i].get<double>());
284 }
285
286 if (values.empty())
287 {
288 r.errorMessage = "JSON contains neither 'parameters' object nor 'values' array";
289 return r;
290 }
291
292 const bool v3 = isCharterV3 (doc) || hasV3Parameters (values);
293 applyValues (v3 ? values : upgradeV2Values (values), apvts, lockedIds, resetUnspecified);
294
295 r.success = true;
296 if (r.presetName.isEmpty())
297 r.presetName = "Preset";
298 return r;
299 }

310 // Sérialise l'état actuel de l'APVTS au format hel.json : tous les paramètres, normalisés.
311 // Utilisé par le bouton "Save preset" et par l'affinage (preset courant).
312 inline juce::String saveToJsonString (juce::AudioProcessorValueTreeState& apvts,
313 const juce::String& presetName, const juce::String& generatedBy)
314 {
315 const bool dx7 = apvts.getRawParameterValue ("engine") != nullptr
316 && apvts.getRawParameterValue ("engine")->load() > 0.5f;
317
318 nlohmann::json doc;
319 doc["metadata"] = {
320 { "name", presetName.toStdString() },
321 { "generated_by", generatedBy.toStdString() },
322 { "charter_version", std::string (HarmoniaCharter::version) },
323 { "engine", dx7 ? "dx7" : "analog" }
324 };
325
326 nlohmann::json paramsObj = nlohmann::json::object();
327 for (const auto& spec : HarmoniaCharter::params)
328 if (auto* p = apvts.getParameter (HarmoniaParams::toJuce (spec.id)))
329 paramsObj[std::string (spec.id)] = p->getValue(); // normalisé [0..1]
330
331 doc["parameters"] = paramsObj;
332 return juce::String (doc.dump (4));
333 }
