21 // Vrai si le preset suit la charte v3 (metadata.charter_version "3...") : ses valeurs sont
22 // normalisées sur les plages v3. Sinon c'est un preset v1/v2 (modèles v1/v2, anciens fichiers),
23 // normalisé sur les 20 plages v2.
24 inline bool isCharterV3 (const nlohmann::json& doc)
25 {
26 if (! doc.contains ("metadata") || ! doc["metadata"].is_object())
27 return false;
28 const auto& m = doc["metadata"];
29 return m.contains ("charter_version") && m["charter_version"].is_string()
30 && m["charter_version"].get<std::string>().rfind ("3", 0) == 0;
31 }

33 // Vrai si les valeurs contiennent un paramètre apparu en v3 (engine, unison_voices, dx7_...) : seul
34 // indice quand le preset arrive sans metadata, comme une entrée d'historique du backend (un nom et
35 // ses parameters). Un preset v1/v2 n'a que les 20 paramètres v2 ; un id inconnu ne compte pas.
36 inline bool hasV3Parameters (const std::map<juce::String, float>& values)
37 {
38 for (const auto& entry : values)
39 {
40 const auto id = entry.first.toStdString();
41 if (HarmoniaCharter::indexOf (id) >= 0
42 && std::find (HarmoniaCharter::v2Ids.begin(), HarmoniaCharter::v2Ids.end(), id) == HarmoniaCharter::v2Ids.end())
43 return true;
44 }
45 return false;
46 }

48 // Applique des valeurs normalisées { id -> [0..1] } sur l'APVTS. Un preset v3 s'applique tel quel.
49 // Un preset v1/v2 (normalisé sur les 20 plages v2) passe par les règles de la charte
50 // (HarmoniaCharter::upgradeFromV2) ; les paramètres apparus en v3 reprennent leur valeur neutre,
51 // pour que le son reste celui du preset, et le moteur passe en analogique.
52 // lockedIds : paramètres verrouillés, ils ne sont jamais modifiés.
53 // Renvoie le nombre de paramètres du preset reconnus (verrouillés compris).
54 inline int applyValues (const std::map<juce::String, float>& values, bool charterV3,
55 juce::AudioProcessorValueTreeState& apvts,
56 const juce::StringArray& lockedIds)
57 {
58 auto applyNorm = [&apvts, &lockedIds] (const juce::String& id, float normValue)
59 {
60 if (lockedIds.contains (id))
61 return; // verrouille : on ne touche pas
62
63 if (auto* p = apvts.getParameter (id))
64 {
65 normValue = juce::jlimit (0.0f, 1.0f, normValue);
66 p->beginChangeGesture();
67 p->setValueNotifyingHost (normValue);
68 p->endChangeGesture();
69 }
70 };
71
72 int recognised = 0;
73
74 if (charterV3)
75 {
76 for (const auto& [id, value] : values)
77 {
78 if (apvts.getParameter (id) != nullptr)
79 ++recognised;
80 applyNorm (id, value);
81 }
82 return recognised;
83 }
84
85 // preset v1/v2 : valeurs normalisées sur les plages v2 -> valeurs physiques v3
86 std::array<float, HarmoniaCharter::v2Ids.size()> v2 {};
87 std::array<bool, HarmoniaCharter::v2Ids.size()> present {};
88 for (size_t i = 0; i < v2.size(); ++i)
89 {
90 const auto it = values.find (HarmoniaParamsV3::toJuce (HarmoniaCharter::v2Ids[i]));
91 present[i] = it != values.end();
92 v2[i] = present[i] ? juce::jlimit (0.0f, 1.0f, it->second) : 0.0f;
93 recognised += present[i] ? 1 : 0;
94 }
95 for (size_t i = HarmoniaCharter::analogBegin; i < HarmoniaCharter::analogEnd; ++i)
96 {
97 const auto& p = HarmoniaCharter::params[i];
98 const auto id = HarmoniaParamsV3::toJuce (p.id);
99 if (p.v2.source < 0)
100 applyNorm (id, HarmoniaCharter::toNormalized (p, p.defaultValue));
101 else if (present[(size_t) p.v2.source])
102 applyNorm (id, HarmoniaCharter::toNormalized (p, HarmoniaCharter::upgradeFromV2Normalized (p, v2)));
103 // absent du preset : le paramètre garde sa valeur
104 }
105 applyNorm ("engine", 0.0f); // analogique
106 return recognised;
107 }

109 // Charge une entrée de l'historique du backend : un nom et { id -> valeur normalisée [0..1] },
110 // sans metadata. Le format (v3, ou v1/v2 à convertir) se lit sur les ids (hasV3Parameters).
111 // Les ids verrouillés ou inconnus sont ignorés ; une valeur hors de [0, 1] refuse l'entrée entière.
112 inline LoadResult loadFromParameters (const std::map<juce::String, float>& parameters,
113 const juce::String& presetName,
114 juce::AudioProcessorValueTreeState& apvts,
115 const juce::StringArray& lockedIds = {})
116 {
117 LoadResult r;
118
119 // Valider avant de toucher au moindre paramètre
120 for (const auto& [id, value] : parameters)
121 {
122 if (! std::isfinite (value) || value < 0.0f || value > 1.0f)

124 r.errorMessage = "Invalid preset: value out of range";
125 return r;

127 }

134

137 return r;

140 // Charge un fichier hel.json (schéma : metadata + parameters{name:value} + values[])
141 // et applique les valeurs normalisées sur l'APVTS.
142 // Accepte deux formes :
143 // 1. parameters: { "osc_1_waveform": 0.5, ... } (recommandée)
144 // 2. values: [0.5, 0.3, ...] (fallback v2 ; ordre = HarmoniaParams::orderedIds())
145 // Un preset v3 (metadata.charter_version "3...", ou des paramètres apparus en v3) s'applique
146 // tel quel ; un preset v1/v2 est converti (voir applyValues).
147 //
148 // lockedIds : paramètres verrouillés, ils ne sont jamais modifiés.
149 inline LoadResult loadFromJsonString (const juce::String& jsonText,
150 juce::AudioProcessorValueTreeState& apvts,
151 const juce::StringArray& lockedIds = {})
152 {
153 LoadResult r;
154
155 nlohmann::json doc;
156 try { doc = nlohmann::json::parse (jsonText.toStdString()); }
157 catch (const std::exception& e)
158 {
159 r.errorMessage = juce::String ("Invalid JSON: ") + e.what();
160 return r;
161 }
162
163
164 if (doc.contains ("metadata")
165 && doc["metadata"].is_object()
166 && doc["metadata"].contains ("name")
167 && doc["metadata"]["name"].is_string())
168 {
169 r.presetName = juce::String (doc["metadata"]["name"].get<std::string>());
170 }
171
172 // Valeurs normalisées du preset, par identifiant.
173 std::map<juce::String, float> values;
174
175 if (doc.contains ("parameters") && doc["parameters"].is_object())
176 {
177 for (auto it = doc["parameters"].begin(); it != doc["parameters"].end(); ++it)
178 if (it.value().is_number())
179 values[juce::String (it.key())] = (float) it.value().get<double>();
180 }
181 else if (doc.contains ("values") && doc["values"].is_array())
182 {
183 const auto& ids = HarmoniaParams::orderedIds();
184 const auto& arr = doc["values"];
185
186 // Le tableau doit correspondre exactement aux paramètres attendus.
187 if (arr.size() != ids.size())
188 {
189 r.errorMessage = "Invalid preset: incorrect number of values";
190 return r;
191 }
192
193 // Valider toutes les valeurs avant de modifier le moindre paramètre.
194 for (const auto& value : arr)
195 {
196 if (!value.is_number())
197 {
198 r.errorMessage = "Invalid preset: 'values' must contain only numbers";
199 return r;
200 }
201
202 const double number = value.get<double>();
203
204 if (!std::isfinite(number)
205 || number < 0.0
206 || number > 1.0)
207 {
208 r.errorMessage = "Invalid preset: value out of range";
209 return r;
210 }
211 }
212
213 for (size_t i = 0; i < ids.size(); ++i)
214 values[ids[i]] = static_cast<float> (arr[i].get<double>());
215 }
216
217 const bool applied = ! values.empty();
218
219 if (applied)
220 applyValues (values, isCharterV3 (doc) || hasV3Parameters (values), apvts, lockedIds);
221
222 if (! applied)
223 {
224 r.errorMessage = "JSON contains neither 'parameters' object nor 'values' array";
225 return r;
226 }
227
228 r.success = true;
229 if (r.presetName.isEmpty())
230 r.presetName = "Preset";
231 return r;
232 }

243 // Sérialise l'état actuel de l'APVTS au format hel.json, charte v3 : tous les paramètres,
244 // normalisés. Utilisé par le bouton "Save preset" et par l'affinage (preset courant).
245 inline juce::String saveToJsonString (juce::AudioProcessorValueTreeState& apvts,
246 const juce::String& presetName, const juce::String& generatedBy)
247 {
248 const bool dx7 = apvts.getRawParameterValue ("engine") != nullptr
249 && apvts.getRawParameterValue ("engine")->load() > 0.5f;
250
251 nlohmann::json doc;
252 doc["metadata"] = {
253 { "name", presetName.toStdString() },
254 { "generated_by", generatedBy.toStdString() },
255 { "charter_version", std::string (HarmoniaCharter::version) },
256 { "engine", dx7 ? "dx7" : "analog" }
257 };
258
259 nlohmann::json paramsObj = nlohmann::json::object();
260 for (const auto& spec : HarmoniaCharter::params)
261 if (auto* p = apvts.getParameter (HarmoniaParamsV3::toJuce (spec.id)))
262 paramsObj[std::string (spec.id)] = p->getValue(); // normalisé [0..1]
263
264 doc["parameters"] = paramsObj;
265 return juce::String (doc.dump (4));
266 }
