390 inline float snap (const Param& p, float value) noexcept
391 {
392 value = std::clamp (value, p.min, p.max);
393 if (p.kind == Kind::Bool)
394 value = value - p.min >= 0.5f * (p.max - p.min) ? p.max : p.min;
395 else if (p.kind == Kind::Int || p.kind == Kind::Choice)
396 value = std::nearbyint (value); // default rounding mode: to nearest, halves to even
397 else if (p.interval > 0.0f)
398 value = p.min + p.interval * std::floor ((value - p.min) / p.interval + 0.5f);
399 return std::clamp (value, p.min, p.max);
400 }
401
403 inline float toNormalized (const Param& p, float value) noexcept
404 {
405 const float proportion = (snap (p, value) - p.min) / (p.max - p.min);
406 return (p.skew != 1.0f && proportion > 0.0f) ? std::pow (proportion, p.skew) : proportion;
407 }
408
410 inline float fromNormalized (const Param& p, float normalized) noexcept
411 {
412 float proportion = std::clamp (normalized, 0.0f, 1.0f);
413 if (p.skew != 1.0f && proportion > 0.0f)
414 proportion = std::exp (std::log (proportion) / p.skew);
415 return snap (p, p.min + (p.max - p.min) * proportion);
416 }
