213 juce::Rectangle<float> cellRect (int i) const
214 {
215 const auto r = boxBounds().toFloat().reduced (1.0f);
216 const int cols = columns > 0 ? juce::jmin (columns, numChoices) : numChoices;
217 const int rows = (numChoices + cols - 1) / juce::jmax (1, cols);
218
219 const float cellW = r.getWidth() / (float) juce::jmax (1, cols);
220 const float cellH = r.getHeight() / (float) juce::jmax (1, rows);
221
222 return { r.getX() + (float) (i % cols) * cellW, r.getY() + (float) (i / cols) * cellH, cellW, cellH };
223 }
224
225 int indexAt (juce::Point<float> p) const
226 {
227 if (numChoices <= 0) return -1;
228 if (isInLockStrip (p)) return -1;
229
230 for (int i = 0; i < numChoices; ++i)
231 if (cellRect (i).expanded (1.0f).contains (p))
232 return i;
233
234 return -1;
235 }
