163 void applyHex()
164 {
165 auto t = hexEditor.getText().trim().removeCharacters ("#");
166
167 if (t.length() != 6 && t.length() != 3)
168 return; // pas encore un code valide
169
170 if (t.length() == 3) // #0af -> #00aaff
171 t = t.substring (0, 1) + t.substring (0, 1)
172 + t.substring (1, 2) + t.substring (1, 2)
173 + t.substring (2, 3) + t.substring (2, 3);
174
175 // déclenche changeListenerCallback -> onColour
176 selector.setCurrentColour (juce::Colour::fromString ("ff" + t));
177 }
