19 void drawButtonBackground(juce::Graphics& g,
20 juce::Button& button,
21 const juce::Colour&,
22 bool isMouseOver,
23 bool isDown) override
24 {
25 if (m_style == BtnStyle::Ghost)
26 return;
27
28 // Souris OU focus clavier (Tab) : même retour visuel
29 const bool isHighlighted = isMouseOver || button.hasKeyboardFocus(true);
30
31 // 4 px margin so the glow is not clipped
32 auto area = button.getLocalBounds().toFloat().reduced(4.f);
33 if (isDown)
34 area = area.reduced(1.f);
35
36 const float r = area.getHeight() * 0.5f; // pill
37
38 // Glass fill
39 g.setColour(juce::Colours::white.withAlpha(isDown ? 0.10f
40 : isHighlighted ? 0.07f : 0.035f));
41 g.fillRoundedRectangle(area, r);
42
43 // Soft glow on hover / focus
44 if (isHighlighted || isDown)
45 {
46 g.setColour(HarmoniaColours::waveCyan.withAlpha(0.10f));
47 g.drawRoundedRectangle(area.expanded(1.5f), r + 1.5f, 5.f);
48
49 g.setColour(HarmoniaColours::waveCyan.withAlpha(0.16f));
50 g.drawRoundedRectangle(area.expanded(0.5f), r + 0.5f, 2.f);
51 }
52
53 // Gradient border cyan -> indigo
54 const float a = isHighlighted || isDown ? 1.0f : 0.65f;
55
56 juce::ColourGradient border(
57 HarmoniaColours::waveCyan.withAlpha(a), area.getX(), area.getY(),
58 HarmoniaColours::waveIndigo.withAlpha(a), area.getRight(), area.getBottom(),
59 false);
60 g.setGradientFill(border);
61 g.drawRoundedRectangle(area, r, 1.3f);
62 }

64 void drawButtonText(juce::Graphics& g,
65 juce::TextButton& button,
66 bool isMouseOver,
67 bool /*isDown*/) override
68 {
69 const bool isHighlighted = isMouseOver || button.hasKeyboardFocus(true);
70 const auto b = button.getLocalBounds();
71
72 if (m_style == BtnStyle::Ghost)
73 {
74 const juce::Font font("Inter", 13.f, juce::Font::plain);
75 g.setFont(font);
76
77 // shadow: stays readable even over a bright wave
78 g.setColour(juce::Colours::black.withAlpha(0.45f));
79 g.drawText(button.getButtonText(), b.translated(0, 1),
80 juce::Justification::centred, false);
81
82 g.setColour(juce::Colours::white.withAlpha(isHighlighted ? 1.0f : 0.80f));
83 g.drawText(button.getButtonText(), b, juce::Justification::centred, false);
84
85 if (isHighlighted)
86 {
87 juce::GlyphArrangement ga;
88 ga.addLineOfText(font, button.getButtonText(), 0.f, 0.f);
89 const float tw = ga.getBoundingBox(0, -1, true).getWidth();
90 g.fillRect((float) b.getCentreX() - tw * 0.5f,
91 (float) b.getCentreY() + 9.f, tw, 1.f);
92 }
93 return;
94 }
95
96 g.setColour(HarmoniaColours::textPrimary.withAlpha(isHighlighted ? 1.0f : 0.90f));
97 g.setFont(juce::Font("Inter", 14.f, juce::Font::plain)
98 .withExtraKerningFactor(0.04f));
99 g.drawFittedText(button.getButtonText(), b.reduced(10, 0),
100 juce::Justification::centred, 1);
101 }
