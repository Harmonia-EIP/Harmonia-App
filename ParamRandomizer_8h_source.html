10//==============================================================================
11// Génère un patch aléatoire MAIS jouable.
12//
13// Les plages musicales (en unités réelles) sont écrites dans randomize(), mais
14// TOUT passe par le charter :
15// - toute plage est rognée à [min, max] du charter (+ DBG si elle débordait)
16// - la valeur finale est arrondie par HarmoniaCharter::snap (int / choice / pas)
17// - le nombre de choix (waveforms, filter type) vient de numChoices
18// - la valeur "éteinte" des tirages sparse est le min du charter
19// - la normalisation (skew compris) est faite par le paramètre lui-même
20//
21// Si le charter change une borne ou une unité, rien ne casse : au pire une plage
22// est rognée et tu as un message DBG qui te dit laquelle ajuster.
23//
24// Paramètres ignorés : `lockedIds` et `excludedIds`. Thread message uniquement.
25// Ne touche PAS à "engine".
26//==============================================================================

29 namespace A = HarmoniaCharter::analog;

45 inline int randomize (juce::AudioProcessorValueTreeState& apvts,
46 const juce::StringArray& lockedIds,
47 const juce::StringArray& excludedIds = {})
48 {
49 using namespace HarmoniaParams;
50
51 juce::Random rng;
52 int changed = 0;
53
54 auto isFree = [&] (const juce::String& id)
55 {
56 return ! lockedIds.contains (id) && ! excludedIds.contains (id);
57 };
58
59 // Envoie une valeur réelle : snap par le charter, normalisation par le paramètre.
60 auto setReal = [&] (A::Index idx, float value)
61 {
62 const auto id = IDs::idOf (idx);
63 if (! isFree (id))
64 return;
65
66 if (auto* p = apvts.getParameter (id))
67 {
68 const auto& s = spec (idx);
69
70 if (value < s.min || value > s.max)
71 DBG ("ParamRandomizer: " << id << " = " << value << " hors charter ["
72 << s.min << ", " << s.max << "] -> borne");
73
74 const float v = HarmoniaCharter::snap (s, value); // clamp + arrondi
75
76 p->beginChangeGesture();
77 p->setValueNotifyingHost (p->convertTo0to1 (v));
78 p->endChangeGesture();
79 ++changed;
80 }
81 };
82
83 // Plage rognée au charter.
84 auto clampLo = [] (A::Index i, float v) { return std::max (v, spec (i).min); };
85 auto clampHi = [] (A::Index i, float v) { return std::min (v, spec (i).max); };
86
87 auto lin = [&] (A::Index i, float lo, float hi)
88 {
89 lo = clampLo (i, lo);
90 hi = clampHi (i, hi);
91 return lo + (hi - lo) * rng.nextFloat();
92 };
93
94 // log : bias > 1 favorise les petites valeurs
95 auto logv = [&] (A::Index i, float lo, float hi, float bias = 1.0f)
96 {
97 lo = std::max (clampLo (i, lo), 1.0e-3f);
98 hi = clampHi (i, hi);
99 if (hi <= lo)
100 return lo;
101 return lo * std::pow (hi / lo, std::pow (rng.nextFloat(), bias));
102 };
103
104 // `chanceOff` de rester au min du charter, sinon une valeur dans [lo, hi]
105 auto sparse = [&] (A::Index i, float chanceOff, float lo, float hi)
106 {
107 return rng.nextFloat() < chanceOff ? spec (i).min : lin (i, lo, hi);
108 };
109
110 auto choice = [&] (A::Index i)
111 {
112 return (float) rng.nextInt (std::max (1, spec (i).numChoices));
113 };
114
115 // ---------- paramètres apparus en v3 : retour aux défauts ----------
116 for (std::size_t i = numLegacy; i < (std::size_t) A::count; ++i)
117 {
118 const auto id = IDs::idOf ((A::Index) i);
119
120 if (! isFree (id))
121 continue;
122
123 if (auto* p = apvts.getParameter (id))
124 {
125 p->beginChangeGesture();
126 p->setValueNotifyingHost (p->getDefaultValue());
127 p->endChangeGesture();
128 ++changed;
129 }
130 }
131
132 // ---------- filtre : le type d'abord, il conditionne le cutoff ----------
133 int filterType = 0;
134
135 if (isFree (IDs::filterType))
136 {
137 const float r = rng.nextFloat();
138 filterType = r < 0.70f ? 0 : (r < 0.85f ? 1 : 2); // surtout lowpass
139 filterType = std::min (filterType, spec (A::filter_type).numChoices - 1);
140 setReal (A::filter_type, (float) filterType);
141 }
142 else if (auto* raw = apvts.getRawParameterValue (IDs::filterType))
143 {
144 filterType = (int) raw->load(); // verrouillé : on garde le sien
145 }
146
147 switch (filterType)
148 {
149 case 1: setReal (A::filter_cutoff, logv (A::filter_cutoff, 300.0f, 6000.0f)); break; // bandpass
150 case 2: setReal (A::filter_cutoff, logv (A::filter_cutoff, 60.0f, 2500.0f)); break; // highpass
151 default: setReal (A::filter_cutoff, logv (A::filter_cutoff, 400.0f, 14000.0f)); break; // lowpass
152 }
153
154 setReal (A::filter_resonance, lin (A::filter_resonance, 0.05f, 0.70f));
155
156 // ---------- oscillateurs ----------
157 setReal (A::osc_1_waveform, choice (A::osc_1_waveform));
158 setReal (A::osc_2_waveform, choice (A::osc_2_waveform));
159 setReal (A::osc_mix, lin (A::osc_mix, 0.10f, 0.90f));
160 setReal (A::osc_2_detune, sparse (A::osc_2_detune, 0.20f, 2.0f, 40.0f)); // cents
161 setReal (A::noise_level, sparse (A::noise_level, 0.70f, 0.02f, 0.20f));
162
163 // ---------- enveloppe d'amplitude (ms) ----------
164 setReal (A::amp_attack, logv (A::amp_attack, 1.0f, 600.0f, 2.0f)); // plutôt rapide
165 setReal (A::amp_decay, logv (A::amp_decay, 60.0f, 2500.0f));
166 setReal (A::amp_sustain, lin (A::amp_sustain, 0.35f, 1.0f)); // jamais muet
167 setReal (A::amp_release, logv (A::amp_release, 60.0f, 3000.0f, 1.3f));
168
169 // ---------- enveloppe de filtre ----------
170 setReal (A::filter_env_amount, lin (A::filter_env_amount, -3.0f, 6.0f)); // octaves
171 setReal (A::filter_env_decay, logv (A::filter_env_decay, 60.0f, 2500.0f));
172
173 // ---------- LFO / modulations ----------
174 setReal (A::lfo_rate, logv (A::lfo_rate, 0.2f, 12.0f));
175 setReal (A::lfo_to_pitch, sparse (A::lfo_to_pitch, 0.70f, 5.0f, 60.0f)); // cents
176 setReal (A::lfo_to_cutoff, sparse (A::lfo_to_cutoff, 0.50f, 0.2f, 2.0f)); // octaves
177 setReal (A::velocity_to_filter, sparse (A::velocity_to_filter, 0.40f, 0.05f, 0.70f));
178
179 // ---------- FX ----------
180 setReal (A::distortion_mix, sparse (A::distortion_mix, 0.60f, 0.05f, 0.50f));
181 setReal (A::reverb_mix, lin (A::reverb_mix, 0.05f, 0.55f));
182
183 return changed;
184 }
