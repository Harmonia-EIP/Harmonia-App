47 int lowbits = phase & ((1 << SHIFT) - 1);
48#ifdef SIN_DELTA
49 int phase_int = (phase >> (SHIFT - 1)) & ((SIN_N_SAMPLES - 1) << 1);
50 int dy = sintab[phase_int];
