105 void paint (juce::Graphics& g) override
106 {
107 const float ease = 1.f - std::pow (1.f - appear, 3.f);
108 const auto accent = accentFor (type);
109 const auto card = getLocalBounds().toFloat().reduced ((float) glowMargin);
110
111 g.beginTransparencyLayer (ease);
112
113 {
114 juce::Graphics::ScopedSaveState ss (g);
115
116 // petit glissement d'entrée, depuis le haut ou le bas selon la position
117 g.addTransform (juce::AffineTransform::translation (0.f, (1.f - ease) * (isBottom (position) ? 8.f : -8.f)));
118
119 // lueur (pré-calculée)
120 g.drawImageAt (glowImage, 0, 0);
121
122 // fond
123 g.setGradientFill (juce::ColourGradient (bgTop(), card.getX(), card.getY(),
124 bgBottom(), card.getX(), card.getBottom(), false));
125 g.fillRoundedRectangle (card, cornerRadius);
126
127 // bordure d'accent
128 g.setColour (accent.withAlpha (0.55f));
129 g.drawRoundedRectangle (card.reduced (0.5f), cornerRadius, 1.2f);
130
131 // icône
132 const float iconD = 34.f;
133 drawIcon (g, type, juce::Rectangle<float> (iconD, iconD)
134 .withCentre ({ card.getX() + padding + iconD * 0.5f, card.getCentreY() }));
135
136 // titre
137 const float textX = card.getX() + textOffset;
138 g.setColour (textMain());
139 g.setFont (juce::Font ("Space Grotesk", 14.5f, juce::Font::bold));
140 g.drawText (title, juce::Rectangle<float> (textX, card.getY() + padding, (float) textWidth, titleHeight),
141 juce::Justification::centredLeft, true);
142
143 // message
144 if (message.isNotEmpty())
145 messageLayout.draw (g, { textX, card.getY() + padding + titleHeight + 3.f,
146 (float) textWidth, messageLayout.getHeight() });
147
148 // barre de temps restant, clippée aux coins arrondis
149 {
150 juce::Path clip;
151 clip.addRoundedRectangle (card, cornerRadius);
152 g.reduceClipRegion (clip);
153
154 const float remaining = juce::jlimit (0.f, 1.f, 1.f - (float) (elapsed / (double) durationMs));
155 auto bar = card.withTop (card.getBottom() - 2.5f).withWidth (card.getWidth() * remaining);
156
157 g.setGradientFill (juce::ColourGradient (accent.withAlpha (0.95f), bar.getX(), 0.f,
158 accent.withAlpha (0.35f), card.getRight(), 0.f, false));
159 g.fillRect (bar);
160 }
161 }
162
163 g.endTransparencyLayer();

234 void buildLayoutAndCache()
235 {
236 float messageH = 0.f;
237
238 if (message.isNotEmpty())
239 {
240 juce::AttributedString s;
241 s.setJustification (juce::Justification::topLeft);
242 s.append (message, juce::Font ("Inter", 12.f, juce::Font::plain), textMain().withAlpha (0.72f));
243 messageLayout.createLayout (s, (float) textWidth);
244 messageH = messageLayout.getHeight() + 3.f;
245 }
246
247 const float cardH = juce::jmax (padding * 2.f + 34.f,
248 padding * 2.f + titleHeight + messageH + 2.f);
249
250 setSize (cardWidth + glowMargin * 2, (int) std::ceil (cardH) + glowMargin * 2);
251
252 // lueur pré-calculée une fois (le flou à 60 Hz serait inutilement coûteux)
253 glowImage = juce::Image (juce::Image::ARGB, getWidth(), getHeight(), true);
254 juce::Graphics gg (glowImage);
255
256 juce::Path shape;
257 shape.addRoundedRectangle (getLocalBounds().toFloat().reduced ((float) glowMargin), cornerRadius);
258
259 juce::DropShadow (juce::Colours::black.withAlpha (0.55f), 14, { 0, 4 }).drawForPath (gg, shape);
260 juce::DropShadow (accentFor (type).withAlpha (0.28f), 12, { 0, 0 }).drawForPath (gg, shape);

266 static void layoutAll()
267 {
268 std::map<std::pair<juce::Component*, int>, int> used; // décalage cumulé par (racine, position)
269
270 for (auto* t : active())
271 {
272 auto* r = t->root.getComponent();
273
274 if (r == nullptr)
275 continue;
276
277 const auto area = r->getLocalBounds();
278 const int w = t->getWidth(), h = t->getHeight();
279 const int cardH = h - glowMargin * 2;
280 int& offset = used[{ r, (int) t->position }];
281
282 int x;
283 if (isLeft (t->position)) x = edgeGap - glowMargin;
284 else if (isRight (t->position)) x = area.getRight() - edgeGap + glowMargin - w;
285 else x = area.getCentreX() - w / 2;
286
287 int y;
288 if (isTop (t->position)) y = topInset - glowMargin + offset;
289 else if (isBottom (t->position)) y = area.getBottom() - edgeGap + glowMargin - h - offset;
290 else y = area.getCentreY() - h / 2 + offset;
291
292 t->setTopLeftPosition (x, y);
293 offset += cardH + stackGap;
294 }
295 }

302 void timerCallback() override
303 {
304 // la fenêtre racine a disparu avant nous
305 if (root == nullptr)
306 {
307 finish();
308 return;
309 }
310
311 const double now = juce::Time::getMillisecondCounterHiRes();
312 const double dt = juce::jlimit (0.0, 100.0, now - lastTick);
313 lastTick = now;
314
315 if (closing)
316 {
317 appear -= (float) (dt / 300.0);
318
319 if (appear <= 0.f)
320 {
321 appear = 0.f;
322 finish();
323 return;
324 }
325 }
326 else
327 {
328 appear = juce::jmin (1.f, appear + (float) (dt / 220.0));
329
330 if (! isMouseOver (true)) // survol = pause
331 elapsed += dt;
332
333 if (elapsed >= (double) durationMs)
334 closing = true;
335 }
336
337 repaint();
338 }

352 static void drawIcon (juce::Graphics& g, Type t, juce::Rectangle<float> c)
353 {
354 const auto col = accentFor (t);
355
356 g.setColour (col.withAlpha (0.16f));
357 g.fillEllipse (c);
358 g.setColour (col.withAlpha (0.85f));
359 g.drawEllipse (c.reduced (1.f), 1.4f);
360
361 g.setColour (col);
362 const auto m = c.getCentre();
363 const juce::PathStrokeType stroke (2.3f, juce::PathStrokeType::curved, juce::PathStrokeType::rounded);
364 juce::Path p;
365
366 switch (t)
367 {
368 case Type::Error:
369 p.startNewSubPath (m.translated (-5.5f, -5.5f));
370 p.lineTo (m.translated ( 5.5f, 5.5f));
371 p.startNewSubPath (m.translated ( 5.5f, -5.5f));
372 p.lineTo (m.translated (-5.5f, 5.5f));
373 g.strokePath (p, stroke);
374 break;
375
376 case Type::Warning: // "!"
377 p.startNewSubPath (m.translated (0.f, -7.f));
378 p.lineTo (m.translated (0.f, 2.f));
379 g.strokePath (p, stroke);
380 g.fillEllipse (juce::Rectangle<float> (3.8f, 3.8f).withCentre (m.translated (0.f, 7.f)));
381 break;
382
383 case Type::Success: // "✓"
384 p.startNewSubPath (m.translated (-6.f, 0.5f));
385 p.lineTo (m.translated (-1.8f, 4.8f));
386 p.lineTo (m.translated ( 6.5f, -4.8f));
387 g.strokePath (p, stroke);
388 break;
389
390 case Type::Info: // "i"
391 p.startNewSubPath (m.translated (0.f, -1.5f));
392 p.lineTo (m.translated (0.f, 7.f));
393 g.strokePath (p, stroke);
394 g.fillEllipse (juce::Rectangle<float> (3.8f, 3.8f).withCentre (m.translated (0.f, -7.f)));
395 break;
