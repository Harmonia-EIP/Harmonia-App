21// Police de la barre de prompt : UNE seule source de vérité pour le texte saisi
22// et le placeholder, en Generate comme en Refine.
23// La police monospace paraît plus petite que la police normale à hauteur égale :
24// on corrige son échelle pour que la hauteur de lettres soit la même qu'en Generate.
25namespace PromptFont
26{
27 inline constexpr float size = 14.0f; // <- LA taille, pour les deux modes
28
29 // Échelle du mono par rapport à Generate. 1.0 = même valeur numérique (14).
30 // - le texte Refine paraît trop GROS -> baisse (0.94f, 0.90f, ...)
31 // - le texte Refine paraît trop PETIT -> monte (1.04f, 1.08f, ...)
32 inline constexpr float monoScale = 0.90f;
33
34 // Espacement des lettres du mono (proportion de la taille de police).
35 // Une police monospace est naturellement plus "aérée" : on la resserre.
36 // - encore trop espacé -> plus négatif (-0.10f, -0.14f, ...)
37 // - trop serré -> plus proche de 0 (-0.03f, 0.0f)
38 inline constexpr float monoKerning = -0.07f;
39
40 inline juce::Font make (bool mono)
41 {
42 if (! mono)
43 return juce::Font (juce::FontOptions (size));
44
45 return juce::Font (juce::FontOptions (juce::Font::getDefaultMonospacedFontName(),
46 size * monoScale,
47 juce::Font::plain))
48 .withExtraKerningFactor (monoKerning);
49 }
50}
